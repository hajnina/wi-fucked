# Packages installed into the image.
# Kept as one package per line so update.sh and setup_rpi.sh share one source.

# LAN side — the AP is the anchor and must come up without the daemon.
hostapd
dnsmasq
avahi-daemon
# A radio that boots rfkill-soft-blocked (no wpa_supplicant run has ever
# unblocked it) makes hostapd fail nl80211 init with no SSID and no retry —
# hostapd's drop-in unblocks it on every start, see setup_rpi.sh.
rfkill

# Data plane. Without these the appliance cannot do its job, which is why the
# CI capability gate asserts sch_cake and wireguard are actually present.
nftables
iproute2
wireguard-tools

# WAN side
network-manager
modemmanager
usbmuxd
libimobiledevice-utils
ethtool
# Wi-Fi-as-WAN (hal/linux.py LinuxWifi): wlan0 is unmanaged by NetworkManager
# (setup_rpi.sh, ADR-011) so nmcli cannot scan/associate on it — station
# scan/connect/link go through iw + wpa_supplicant directly, with dhclient for
# address assignment. See docs/active-tests.md, "Wi-Fi-as-WAN via iw/wpa_supplicant".
iw
wpasupplicant
isc-dhcp-client

# Runtime
python3
python3-flask
python3-pip
unzip
zip
curl
ca-certificates

# Field debugging (SOP-009). Small, and the alternative is being unable to
# diagnose a device you cannot physically reach.
iperf3
tcpdump
